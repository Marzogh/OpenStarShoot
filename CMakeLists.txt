cmake_minimum_required(VERSION 3.20)
project(OpenStarShoot LANGUAGES C)
include(GNUInstallDirs)
include(CTest)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

find_path(LIBUSB_INCLUDE_DIR NAMES libusb.h PATH_SUFFIXES libusb-1.0 REQUIRED)
find_library(LIBUSB_LIBRARY NAMES usb-1.0 libusb-1.0 REQUIRED)
add_library(oss_libusb UNKNOWN IMPORTED)
set_target_properties(oss_libusb PROPERTIES
    IMPORTED_LOCATION "${LIBUSB_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES "${LIBUSB_INCLUDE_DIR}")

function(oss_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endfunction()

set(OSS_SOURCES src/openstarshoot.c src/platform.c)
foreach(kind IN ITEMS static shared)
    string(TOUPPER "${kind}" library_type)
    add_library(openstarshoot_${kind} ${library_type} ${OSS_SOURCES})
    target_include_directories(openstarshoot_${kind} PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
    target_link_libraries(openstarshoot_${kind} PUBLIC oss_libusb)
    set_target_properties(openstarshoot_${kind} PROPERTIES OUTPUT_NAME openstarshoot)
    oss_warnings(openstarshoot_${kind})
endforeach()
# A Windows static archive must not collide with the DLL's import library.
if(WIN32)
    set_target_properties(openstarshoot_static PROPERTIES OUTPUT_NAME openstarshoot_static)
endif()
target_compile_definitions(openstarshoot_shared PUBLIC OSS_SHARED PRIVATE OSS_BUILDING_LIBRARY)
add_executable(starshoot-diag src/cli.c)
target_link_libraries(starshoot-diag PRIVATE openstarshoot_static)
oss_warnings(starshoot-diag)

install(TARGETS openstarshoot_static openstarshoot_shared starshoot-diag
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/openstarshoot.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(PROGRAMS tools/fits_export.py DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME starshoot-fits-export)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/openstarshoot)

if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    # Existing assertions perform test operations too; keep them in Release builds.
    function(oss_test target)
        oss_warnings(${target})
        target_include_directories(${target} PRIVATE include src)
        target_link_libraries(${target} PRIVATE oss_libusb)
        if(MSVC)
            target_compile_options(${target} PRIVATE /UNDEBUG)
        else()
            target_compile_options(${target} PRIVATE -UNDEBUG)
        endif()
    endfunction()
    foreach(name IN ITEMS protocol binning roi capture_cli cooling cli_roi)
        add_executable(test_${name} tests/test_${name}.c ${OSS_SOURCES})
        target_compile_definitions(test_${name} PRIVATE OSS_TESTING)
        oss_test(test_${name})
    endforeach()
    add_executable(test_open tests/test_open.c src/platform.c)
    oss_test(test_open)
    add_executable(test_cli_errors tests/test_cli_errors.c)
    target_link_libraries(test_cli_errors PRIVATE openstarshoot_static)
    oss_test(test_cli_errors)
    add_executable(test_platform tests/test_platform.c src/platform.c)
    oss_test(test_platform)
    foreach(name IN ITEMS protocol binning roi open cli_errors)
        add_test(NAME ${name} COMMAND test_${name})
    endforeach()
    foreach(name IN ITEMS capture_cli cooling platform)
        add_test(NAME ${name} COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_in_temp.py $<TARGET_FILE:test_${name}>)
    endforeach()
    foreach(kind IN ITEMS static shared)
        add_executable(test_link_${kind} tests/test_link.c)
        target_link_libraries(test_link_${kind} PRIVATE openstarshoot_${kind})
        oss_test(test_link_${kind})
        add_test(NAME link_${kind} COMMAND test_link_${kind})
    endforeach()
    option(OSS_TEST_FITS "Run the existing NumPy/Astropy offline FITS regressions" ON)
    if(OSS_TEST_FITS)
        execute_process(COMMAND ${Python3_EXECUTABLE} -c "import numpy; import astropy"
            RESULT_VARIABLE fits_dependencies)
        if(NOT fits_dependencies EQUAL 0)
            message(FATAL_ERROR "FITS tests require NumPy and Astropy. Install them or configure with -DOSS_TEST_FITS=OFF for C tests only.")
        endif()
        add_test(NAME fits COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_fits.py)
        add_test(NAME fits_roi COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_fits_roi.py
            --cli $<TARGET_FILE:test_cli_roi>)
    endif()
    get_property(offline_tests DIRECTORY PROPERTY TESTS)
    set_tests_properties(${offline_tests} PROPERTIES TIMEOUT 120
        ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1;PYTHONUNBUFFERED=1")
endif()
