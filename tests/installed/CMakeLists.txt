cmake_minimum_required(VERSION 3.20)
project(OpenStarShootInstalledTest LANGUAGES C)
include(CTest)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_path(OSS_INCLUDE_DIR openstarshoot.h REQUIRED)
find_path(LIBUSB_INCLUDE_DIR libusb.h PATH_SUFFIXES libusb-1.0 REQUIRED)
find_library(LIBUSB_LIBRARY NAMES usb-1.0 libusb-1.0 REQUIRED)
if(WIN32)
    find_library(OSS_STATIC_LIBRARY NAMES openstarshoot_static REQUIRED)
    find_library(OSS_SHARED_LIBRARY NAMES openstarshoot REQUIRED)
else()
    find_file(OSS_STATIC_LIBRARY NAMES libopenstarshoot.a PATH_SUFFIXES lib lib64 REQUIRED)
    find_library(OSS_SHARED_LIBRARY NAMES openstarshoot REQUIRED)
endif()
foreach(kind IN ITEMS STATIC SHARED)
    add_executable(consumer_${kind} ../test_link.c)
    target_include_directories(consumer_${kind} PRIVATE ${OSS_INCLUDE_DIR} ${LIBUSB_INCLUDE_DIR})
    target_link_libraries(consumer_${kind} PRIVATE ${OSS_${kind}_LIBRARY})
    if(kind STREQUAL "STATIC")
        target_link_libraries(consumer_${kind} PRIVATE ${LIBUSB_LIBRARY})
    else()
        target_compile_definitions(consumer_${kind} PRIVATE OSS_SHARED)
    endif()
    add_test(NAME installed_${kind} COMMAND consumer_${kind})
endforeach()
